import { useState } from "react";
import "./App.css";

export default function App() {
  const [count, setCount] = useState(0);

  return (
    <div className="app">
      <h1>Counter</h1>
      <button onClick={() => setCount(count + 1)}>count is {count}</button>
      <p>Click the button to count.</p>
    </div>
  );
}
